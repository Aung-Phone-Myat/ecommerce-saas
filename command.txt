# Ecommerce SaaS — Commands
# Run from the repository root unless a section says otherwise.

## Setup
cp backend/.env.example backend/.env
cp frontend/.env.example frontend/.env
openssl rand -hex 32              # generate a JWT_SECRET_KEY value

## Backend (run from backend/)
cd backend
uv sync
uv run uvicorn app.main:app --reload --port 8000
uv run pytest
RUN_DB_TESTS=1 uv run pytest tests/test_database.py -v
uv run ruff check .
uv run ruff check migrations --fix
uv run mypy app
uv run alembic revision --autogenerate -m "description"
uv run alembic upgrade head
uv run alembic current
uv run python -c "from app.main import app; print(sorted(app.openapi()['paths']))"

## Frontend (run from frontend/)
cd frontend
npm install
npm ci
npm run dev
npm run lint
npm run build

## Docker Compose (run from repository root)
docker compose up --build
docker compose down
VITE_API_URL=http://localhost:8001 docker compose up --build

## Docker manual (run from repository root)
docker build -t ecommerce-backend ./backend
docker build -t ecommerce-frontend ./frontend --build-arg VITE_API_URL=http://localhost:8000
docker run -d --name ecommerce-backend --env-file ./backend/.env -p 8000:8000 ecommerce-backend
docker run -d --name ecommerce-frontend -p 5173:80 ecommerce-frontend
docker ps
docker logs -f ecommerce-backend
docker stop ecommerce-backend ecommerce-frontend
docker rm ecommerce-backend ecommerce-frontend

## Git / inspection
git status --short
git log --oneline -5 -- backend/app/db/base.py backend/app/db/session.py
git ls-files backend/app/db/
